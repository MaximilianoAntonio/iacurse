   141→function CoursesTab({ authorId }: { authorId: string }) {
   142→  const { data, loading, refetch } = useFetch<{ courses: Course[] }>(
   143→    authorId ? `/api/courses?authorId=${authorId}` : null,
   144→    [authorId]
   145→  );
   146→  const { toast } = useToast();
   147→  const [createOpen, setCreateOpen] = React.useState(false);
   148→  const [editCourse, setEditCourse] = React.useState<Course | null>(null);
   149→  const [editingCourseId, setEditingCourseId] = React.useState<string | null>(null);
   150→  const [form, setForm] = React.useState({ title: "", description: "", color: "sky", icon: "BookOpen" });
   151→
   152→  const handleCreate = async () => {
   153→    if (!form.title.trim()) {
   154→      toast({ title: "Error", description: "El título es obligatorio", variant: "destructive" });
   155→      return;
   156→    }
   157→    try {
   158→      await postJSON("/api/courses", { authorId, ...form });
   159→      setCreateOpen(false);
   160→      setForm({ title: "", description: "", color: "sky", icon: "BookOpen" });
   161→      refetch();
   162→      toast({ title: "Curso creado", description: "El curso sandbox se creó correctamente." });
   163→    } catch {
   164→      toast({ title: "Error", description: "No se pudo crear el curso", variant: "destructive" });
   165→    }
   166→  };
   167→
   168→  const handleDelete = async (courseId: string) => {
   169→    try {
   170→      await fetch(`/api/courses?courseId=${courseId}`, { method: "DELETE" });
   171→      refetch();
   172→      toast({ title: "Curso eliminado" });
   173→    } catch {
   174→      toast({ title: "Error", description: "No se pudo eliminar", variant: "destructive" });
   175→    }
   176→  };
   177→
   178→  const handlePublish = async (course: Course) => {
   179→    try {
   180→      await patchJSON("/api/courses", {
   181→        courseId: course.id,
   182→        status: course.status === "draft" ? "published" : "draft",
   183→      });
   184→      refetch();
   185→      toast({ title: course.status === "draft" ? "Curso publicado" : "Curso en borrador" });
   186→    } catch {
   187→      toast({ title: "Error", description: "No se pudo cambiar el estado", variant: "destructive" });
   188→    }
   189→  };
   190→
   191→  const courses = data?.courses ?? [];
   192→
   193→  if (editingCourseId) {
   194→    return <CourseEditor courseId={editingCourseId} onBack={() => { setEditingCourseId(null); refetch(); }} />;
   195→  }
   196→
   197→  return (
   198→    <div className="space-y-4">
   199→      <div className="flex items-center justify-between">
   200→        <p className="text-sm text-muted-foreground">
   201→          {courses.length} curso{courses.length !== 1 ? "s" : ""} sandbox creado{courses.length !== 1 ? "s" : ""}
   202→        </p>
   203→        <Button size="sm" onClick={() => setCreateOpen(true)} className="bg-[#003366] hover:bg-[#004488]">
   204→          <Plus className="mr-1.5 h-4 w-4" /> Nuevo Curso
   205→        </Button>
   206→      </div>
   207→
   208→      {loading ? (
   209→        <div className="space-y-2">
   210→          {[1, 2, 3].map((i) => (
   211→            <div key={i} className="h-20 animate-pulse rounded-xl bg-muted" />
   212→          ))}
   213→        </div>
   214→      ) : courses.length === 0 ? (
   215→        <Card className="border-dashed">
   216→          <CardContent className="flex flex-col items-center justify-center py-12 text-center">
   217→            <div className="mb-3 flex h-12 w-12 items-center justify-center rounded-full bg-[#003366]/5">
   218→              <BookOpen className="h-5 w-5 text-[#003366]" />
   219→            </div>
   220→            <p className="text-sm font-medium">Sin cursos sandbox</p>
   221→            <p className="text-xs text-muted-foreground">Crea tu primer curso para empezar a agregar contenido</p>
   222→          </CardContent>
   223→        </Card>
   224→      ) : (
   225→        <div className="space-y-3">
   226→          {courses.map((c) => (
   227→            <Card key={c.id} className="transition-shadow hover:shadow-md">
   228→              <CardContent className="flex items-center gap-4 p-4">
   229→                <div className="flex h-11 w-11 shrink-0 items-center justify-center rounded-xl bg-gradient-to-br from-[#003366] to-[#0066AA] text-white shadow-md">
   230→                  <DynamicIcon name={c.icon} className="h-5 w-5" />
   231→                </div>
   232→                <div className="min-w-0 flex-1">
   233→                  <div className="flex items-center gap-2">
   234→                    <h3 className="truncate text-sm font-bold">{c.title}</h3>
   235→                    <Badge
   236→                      variant="outline"
   237→                      className={
   238→                        c.status === "published"
   239→                          ? "border-[#003366]/30 bg-[#003366]/5 text-[#003366]"
   240→                          : "border-amber-300 bg-amber-50 text-amber-700 dark:border-amber-800 dark:bg-amber-950 dark:text-amber-400"
   241→                      }
   242→                    >
   243→                      {c.status === "published" ? "Publicado" : "Borrador"}
   244→                    </Badge>
   245→                  </div>
   246→                  <p className="truncate text-xs text-muted-foreground">
   247→                    {c.description || "Sin descripción"} · {c.unitCount} unidad{c.unitCount !== 1 ? "s" : ""}
   248→                  </p>
   249→                </div>
   250→                <div className="flex shrink-0 items-center gap-1.5">
   251→                  <Button
   252→                    variant="default"
   253→                    size="sm"
   254→                    className="h-8 text-xs bg-[#003366] hover:bg-[#004488]"
   255→                    onClick={() => setEditingCourseId(c.id)}
   256→                  >
   257→                    <BookOpen className="mr-1 h-3.5 w-3.5" /> Editar contenido
   258→                  </Button>
   259→                  <Button
   260→                    variant="ghost"
   261→                    size="sm"
   262→                    className="h-8 text-xs"
   263→                    onClick={() => handlePublish(c)}
   264→                  >
   265→                    {c.status === "draft" ? "Publicar" : "Despublicar"}
   266→                  </Button>
   267→                  <Button
   268→                    variant="ghost"
   269→                    size="sm"
   270→                    className="h-8 text-xs"
   271→                    onClick={() => { setEditCourse(c); setForm({ title: c.title, description: c.description, color: c.color, icon: c.icon }); }}
   272→                  >
   273→                    <Edit2 className="h-3.5 w-3.5" />
   274→                  </Button>
   275→                  <Button
   276→                    variant="ghost"
   277→                    size="sm"
   278→                    className="h-8 text-xs text-rose-600 hover:bg-rose-50"
   279→                    onClick={() => handleDelete(c.id)}
   280→                  >
   281→                    <Trash2 className="h-3.5 w-3.5" />
   282→                  </Button>
   283→                </div>
   284→              </CardContent>
   285→            </Card>
   286→          ))}
   287→        </div>
   288→      )}
   289→
   290→      {/* Dialog crear/editar */}
   291→      <Dialog open={createOpen || editCourse !== null} onOpenChange={(o) => { if (!o) { setCreateOpen(false); setEditCourse(null); } }}>
   292→        <DialogContent className="max-w-md">
   293→          <DialogHeader>
   294→            <DialogTitle>{editCourse ? "Editar curso" : "Crear curso sandbox"}</DialogTitle>
   295→            <DialogDescription>
   296→              Los cursos sandbox son editables y pueden mezclarse con los cursos existentes.
   297→            </DialogDescription>
   298→          </DialogHeader>
   299→          <div className="space-y-3 py-2">
   300→            <div>
   301→              <Label className="text-xs font-semibold">Título</Label>
   302→              <Input value={form.title} onChange={(e) => setForm({ ...form, title: e.target.value })} placeholder="Ej: Introducción a la Electrónica Médica" className="mt-1" />
   303→            </div>
   304→            <div>
   305→              <Label className="text-xs font-semibold">Descripción</Label>
   306→              <Textarea value={form.description} onChange={(e) => setForm({ ...form, description: e.target.value })} placeholder="Breve descripción del curso" className="mt-1 min-h-[70px]" />
   307→            </div>
   308→            <div className="grid grid-cols-2 gap-3">
   309→              <div>
   310→                <Label className="text-xs font-semibold">Color</Label>
   311→                <Input value={form.color} onChange={(e) => setForm({ ...form, color: e.target.value })} placeholder="sky, amber, rose..." className="mt-1" />
   312→              </div>
   313→              <div>
   314→                <Label className="text-xs font-semibold">Icono</Label>
   315→                <Input value={form.icon} onChange={(e) => setForm({ ...form, icon: e.target.value })} placeholder="BookOpen, HeartPulse..." className="mt-1" />
   316→              </div>
   317→            </div>
   318→          </div>
   319→          <DialogFooter>
   320→            <Button variant="outline" size="sm" onClick={() => { setCreateOpen(false); setEditCourse(null); }}>Cancelar</Button>
   321→            <Button
   322→              size="sm"
   323→              onClick={async () => {
   324→                if (editCourse) {
   325→                  try {
   326→                    await patchJSON("/api/courses", { courseId: editCourse.id, ...form });
   327→                    refetch();
   328→                    setEditCourse(null);
   329→                    toast({ title: "Curso actualizado" });
   330→                  } catch {
   331→                    toast({ title: "Error", variant: "destructive" });
   332→                  }
   333→                } else {
   334→                  handleCreate();
   335→                }
   336→              }}
   337→              className="bg-[#003366] hover:bg-[#004488]"
   338→            >
   339→              <Save className="mr-1.5 h-3.5 w-3.5" />
   340→              {editCourse ? "Guardar" : "Crear"}
   341→            </Button>
   342→          </DialogFooter>
   343→        </DialogContent>
   344→      </Dialog>
   345→    </div>
   346→  );
   347→}

   349→// ============ QUESTIONS TAB ============
   350→
   351→function QuestionsTab({ authorId }: { authorId: string }) {
   352→  const { data: banksData, refetch: refetchBanks } = useFetch<{ banks: QuestionBank[] }>(
   353→    authorId ? `/api/question-banks?authorId=${authorId}` : null,
   354→    [authorId]
   355→  );
   356→  const { data: questionsData, refetch: refetchQuestions } = useFetch<{ questions: Question[] }>(
   357→    authorId ? `/api/question-banks/questions?authorId=${authorId}` : null,
   358→    [authorId]
   359→  );
   360→  const { toast } = useToast();
   361→  const [createOpen, setCreateOpen] = React.useState(false);
   362→  const [editQ, setEditQ] = React.useState<Question | null>(null);
   363→  const [form, setForm] = React.useState({
   364→    bankId: "",
   365→    type: "multiple_choice",
   366→    title: "",
   367→    prompt: "",
   368→    data: '{"question":"","options":["A)","B)","C)","D)"],"correctIndex":0,"explanation":"","hints":[]}',
   369→    points: 10,
   370→    difficulty: "medium",
   371→    tags: "",
   372→  });
   373→
   374→  const banks = banksData?.banks ?? [];
   375→  const questions = questionsData?.questions ?? [];
   376→
   377→  const handleSave = async () => {
   378→    if (!form.title.trim() || !form.prompt.trim()) {
   379→      toast({ title: "Error", description: "Título y enunciado son obligatorios", variant: "destructive" });
   380→      return;
   381→    }
   382→    try {
   383→      if (editQ) {
   384→        await patchJSON("/api/question-banks/questions", {
   385→          questionId: editQ.id,
   386→          type: form.type,
   387→          title: form.title,
   388→          prompt: form.prompt,
   389→          data: form.data,
   390→          points: Number(form.points),
   391→          difficulty: form.difficulty,
   392→          tags: form.tags,
   393→        });
   394→        toast({ title: "Pregunta actualizada" });
   395→      } else {
   396→        await postJSON("/api/question-banks/questions", {
   397→          bankId: form.bankId || undefined,
   398→          type: form.type,
   399→          title: form.title,
   400→          prompt: form.prompt,
   401→          data: form.data,
   402→          points: Number(form.points),
   403→          difficulty: form.difficulty,
   404→          tags: form.tags,
   405→        });
   406→        toast({ title: "Pregunta creada" });
   407→      }
   408→      setCreateOpen(false);
   409→      setEditQ(null);
   410→      setForm({
   411→        bankId: "",
   412→        type: "multiple_choice",
   413→        title: "",
   414→        prompt: "",
   415→        data: '{"question":"","options":["A)","B)","C)","D)"],"correctIndex":0,"explanation":"","hints":[]}',
   416→        points: 10,
   417→        difficulty: "medium",
   418→        tags: "",
   419→      });
   420→      refetchQuestions();
   421→    } catch {
   422→      toast({ title: "Error", description: "No se pudo guardar", variant: "destructive" });
   423→    }
   424→  };
   425→
   426→  const handleDelete = async (questionId: string) => {
   427→    try {
   428→      await fetch(`/api/question-banks/questions?questionId=${questionId}`, { method: "DELETE" });
   429→      refetchQuestions();
   430→      toast({ title: "Pregunta eliminada" });
   431→    } catch {
   432→      toast({ title: "Error", variant: "destructive" });
   433→    }
   434→  };
   435→
   436→  const handleCreateBank = async () => {
   437→    const name = prompt("Nombre del banco de preguntas:");
   438→    if (!name) return;
   439→    try {
   440→      await postJSON("/api/question-banks", { authorId, name, description: "", category: "general" });
   441→      refetchBanks();
   442→      toast({ title: "Banco creado" });
   443→    } catch {
   444→      toast({ title: "Error", variant: "destructive" });
   445→    }
   446→  };
   447→
   448→  return (
   449→    <div className="space-y-4">
   450→      {/* Bancos de preguntas */}
   451→      <div className="flex items-center justify-between">
   452→        <div>
   453→          <h3 className="text-sm font-bold">Bancos de Preguntas</h3>
   454→          <p className="text-xs text-muted-foreground">{banks.length} banco{banks.length !== 1 ? "s" : ""} · {questions.length} pregunta{questions.length !== 1 ? "s" : ""} total</p>
   455→        </div>
   456→        <Button size="sm" variant="outline" onClick={handleCreateBank}>
   457→          <Plus className="mr-1.5 h-4 w-4" /> Nuevo Banco
   458→        </Button>
   459→      </div>
   460→
   461→      {banks.length > 0 && (
   462→        <div className="flex flex-wrap gap-2">
   463→          {banks.map((b) => (
   464→            <Badge key={b.id} variant="secondary" className="gap-1.5 py-1.5">
   465→              <FileText className="h-3 w-3" />
   466→              {b.name}
   467→              <span className="ml-1 rounded-full bg-muted px-1.5 text-xs">{b.questionCount}</span>
   468→            </Badge>
   469→          ))}
   470→        </div>
   471→      )}
   472→
   473→      {/* Botón crear pregunta */}
   474→      <div className="flex justify-end">
   475→        <Button size="sm" onClick={() => { setEditQ(null); setCreateOpen(true); }} className="bg-[#003366] hover:bg-[#004488]">
   476→          <Plus className="mr-1.5 h-4 w-4" /> Nueva Pregunta
   477→        </Button>
   478→      </div>
   479→
   480→      {/* Lista de preguntas */}
   481→      {questions.length === 0 ? (
   482→        <Card className="border-dashed">
   483→          <CardContent className="flex flex-col items-center justify-center py-12 text-center">
   484→            <div className="mb-3 flex h-12 w-12 items-center justify-center rounded-full bg-[#003366]/5">
   485→              <ListChecks className="h-5 w-5 text-[#003366]" />
   486→            </div>
   487→            <p className="text-sm font-medium">Sin preguntas</p>
   488→            <p className="text-xs text-muted-foreground">Crea preguntas para reutilizarlas en tus cursos</p>
   489→          </CardContent>
   490→        </Card>
   491→      ) : (
   492→        <div className="space-y-2 max-h-[500px] overflow-y-auto">
   493→          {questions.map((q) => (
   494→            <Card key={q.id} className="transition-shadow hover:shadow-sm">
   495→              <CardContent className="flex items-start gap-3 p-3.5">
   496→                <div className="flex h-8 w-8 shrink-0 items-center justify-center rounded-lg bg-[#003366]/5 text-[#003366]">
   497→                  <DynamicIcon name="ListChecks" className="h-4 w-4" />
   498→                </div>
   499→                <div className="min-w-0 flex-1">
   500→                  <div className="flex items-center gap-2">
   501→                    <p className="truncate text-sm font-medium">{q.title}</p>
   502→                    <Badge variant="outline" className="text-xs">{typeLabels[q.type] ?? q.type}</Badge>
   503→                    <Badge variant="secondary" className="text-xs">{q.difficulty}</Badge>
   504→                  </div>
   505→                  <p className="mt-0.5 truncate text-xs text-muted-foreground">{q.prompt}</p>
   506→                  <p className="mt-0.5 text-xs text-muted-foreground">{q.points} pts · {q.tags || "sin tags"}</p>
   507→                </div>
   508→                <div className="flex shrink-0 gap-1">
   509→                  <Button variant="ghost" size="sm" className="h-7 w-7 p-0" onClick={() => {
   510→                    setEditQ(q);
   511→                    setForm({
   512→                      bankId: q.bankId || "",
   513→                      type: q.type,
   514→                      title: q.title,
   515→                      prompt: q.prompt,
   516→                      data: q.data,
   517→                      points: q.points,
   518→                      difficulty: q.difficulty,
   519→                      tags: q.tags || "",
   520→                    });
   521→                    setCreateOpen(true);
   522→                  }}>
   523→                    <Edit2 className="h-3.5 w-3.5" />
   524→                  </Button>
   525→                  <Button variant="ghost" size="sm" className="h-7 w-7 p-0 text-rose-600" onClick={() => handleDelete(q.id)}>
   526→                    <Trash2 className="h-3.5 w-3.5" />
   527→                  </Button>
