   137→const chartColors: Record<string, string> = {

   155→  violet: "bg-violet-100 text-violet-600 dark:bg-violet-950 dark:text-violet-400",
   156→};
   157→
   158→function formatHoursMinutes(min: number): string {
   159→  if (min <= 0) return "0m";
   160→  if (min < 60) return `${min}m`;
   161→  const h = Math.floor(min / 60);
   162→  const m = min % 60;
   163→  return m === 0 ? `${h}h` : `${h}h ${m}m`;
   164→}
   165→
   166→function formatShortDate(iso: string): string {
   167→  const d = new Date(iso);
   168→  return `${d.getDate()}/${d.getMonth() + 1}`;
   169→}
   170→
   171→function unitHex(color: string): string {
   172→  return chartColors[color] ?? chartColors.emerald;
   173→}
   174→
   175→function typeLabel(t: string): string {
   176→  return activityTypeMeta[t as ActivityType]?.label ?? t;
   177→}
   178→
   179→function difficultyLabel(d: string): string {
   180→  return difficultyMeta[d as Difficulty]?.label ?? d;
   181→}
