   121→function formatTimestamp(iso: string): string {
   122→  try {
   123→    const d = new Date(iso);
   124→    return d.toLocaleTimeString("es-CL", {
   125→      hour: "2-digit",
   126→      minute: "2-digit",
   127→    });
   128→  } catch {
   129→    return "";
   130→  }
   131→}

   137→function StarRating({
   138→  value,
   139→  onChange,
   140→}: {
   141→  value: number;
   142→  onChange: (rating: number) => void;
   143→}) {
   144→  const [hover, setHover] = useState(0);
   145→  const display = hover || value;
   146→  return (
   147→    <div
   148→      className="flex items-center gap-0.5"
   149→      role="group"
   150→      aria-label="Calificar respuesta"
   151→    >
   152→      {[1, 2, 3, 4, 5].map((n) => {
   153→        const active = display >= n;
   154→        return (
   155→          <button
   156→            key={n}
   157→            type="button"
   158→            onClick={() => onChange(n)}
   159→            onMouseEnter={() => setHover(n)}
   160→            onMouseLeave={() => setHover(0)}
   161→            onFocus={() => setHover(n)}
   162→            onBlur={() => setHover(0)}
   163→            aria-label={`${n} estrella${n > 1 ? "s" : ""}${
   164→              value === n ? " (seleccionada)" : ""
   165→            }`}
   166→            className="rounded p-0.5 transition hover:scale-110 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-violet-400"
   167→          >
   168→            <Star
   169→              className={cn(
   170→                "h-3.5 w-3.5 transition-colors",
   171→                active
   172→                  ? "fill-amber-400 text-amber-400"
   173→                  : "text-muted-foreground/40 hover:text-amber-300"
   174→              )}
   175→            />
   176→          </button>
   177→        );
   178→      })}
   179→      {value > 0 && (
   180→        <span className="ml-1 text-[10px] font-medium text-muted-foreground">
   181→          {value}/5
   182→        </span>
   183→      )}
   184→    </div>
   185→  );
   186→}

   353→export function TutorView() {
   354→  const currentUser = useAppStore((s) => s.currentUser) as AppUser | null;
   355→  const tutorContextUnit = useAppStore((s) => s.tutorContextUnit);
   356→  const setTutorContext = useAppStore((s) => s.setTutorContext);
   357→  const { toast } = useToast();
   358→
   359→  const userId = currentUser?.id ?? "";
   360→
   361→  // Cargar unidades para el selector de contexto
   362→  const { data: unitsData } = useFetch<{ units: Unit[] }>(
   363→    userId ? `/api/units?userId=${userId}` : null,
   364→    [userId]
   365→  );
   366→
   367→  // Cargar historial del chat
   368→  const {
   369→    data: historyData,
   370→    loading: historyLoading,
   371→    refetch: refetchHistory,
   372→  } = useFetch<{ messages: ChatMessage[] }>(
   373→    userId ? `/api/tutor?userId=${userId}` : null,
   374→    [userId]
   375→  );
   376→
   377→  const units = useMemo(() => unitsData?.units ?? [], [unitsData]);
   378→
   379→  const [messages, setMessages] = useState<ChatMessage[]>([]);
   380→  const [input, setInput] = useState("");
   381→  const [isSending, setIsSending] = useState(false);
   382→  const [ratedIds, setRatedIds] = useState<Set<string>>(new Set());
   383→
   384→  const bottomRef = useRef<HTMLDivElement>(null);
   385→  const textareaRef = useRef<HTMLTextAreaElement>(null);
   386→
   387→  // Sincronizar historial cargado con el estado local
   388→  useEffect(() => {
   389→    if (historyData?.messages) {
   390→      setMessages(historyData.messages);
   391→    }
   392→  }, [historyData]);
   393→
   394→  // Auto-scroll al final cuando llegan mensajes nuevos o mientras se envía
   395→  useEffect(() => {
   396→    bottomRef.current?.scrollIntoView({ behavior: "smooth", block: "end" });
   397→  }, [messages, isSending]);
   398→
   399→  const selectedContext = tutorContextUnit ?? null;
   400→
   401→  const handleSelectContext = (unitTitle: string | null) => {
   402→    setTutorContext(unitTitle);
   403→  };
   404→
   405→  const handleSend = async () => {
   406→    const trimmed = input.trim();
   407→    if (!trimmed || isSending || !userId) return;
   408→
   409→    const userMessage: ChatMessage = {
   410→      id: `temp-user-${Date.now()}`,
   411→      userId,
   412→      role: "user",
   413→      content: trimmed,
   414→      context: selectedContext,
   415→      rating: null,
   416→      createdAt: new Date().toISOString(),
   417→    };
   418→
   419→    setMessages((prev) => [...prev, userMessage]);
   420→    setInput("");
   421→    setIsSending(true);
   422→
   423→    try {
   424→      const res = await postJSON<{
   425→        message: { id: string; role: "assistant"; content: string; createdAt: string };
   426→      }>("/api/tutor", {
   427→        userId,
   428→        message: trimmed,
   429→        context: selectedContext ?? undefined,
   430→      });
   431→
   432→      const assistantMessage: ChatMessage = {
   433→        id: res.message.id,
   434→        userId,
   435→        role: "assistant",
   436→        content: res.message.content,
   437→        context: selectedContext,
   438→        rating: null,
   439→        createdAt: res.message.createdAt,
   440→      };
   441→      setMessages((prev) => [...prev, assistantMessage]);
   442→    } catch (err) {
   443→      const msg = err instanceof Error ? err.message : "No se pudo enviar el mensaje";
   444→      toast({
   445→        title: "Error de conexión",
   446→        description: msg,
   447→        variant: "destructive",
   448→      });
   449→      // Reinsertar el texto del usuario para que no se pierda
   450→      setInput(trimmed);
   451→      // Remover el mensaje optimista fallido
   452→      setMessages((prev) => prev.filter((m) => m.id !== userMessage.id));
   453→    } finally {
   454→      setIsSending(false);
   455→      // Devolver foco al textarea
   456→      requestAnimationFrame(() => textareaRef.current?.focus());
   457→    }
   458→  };
   459→
   460→  const handleKeyDown = (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   461→    if (e.key === "Enter" && !e.shiftKey) {
   462→      e.preventDefault();
   463→      handleSend();
   464→    }
   465→  };
   466→
   467→  const handleRate = async (messageId: string, rating: number) => {
   468→    // Actualización optimista local
   469→    setMessages((prev) =>
   470→      prev.map((m) => (m.id === messageId ? { ...m, rating } : m))
   471→    );
   472→
   473→    const isFirstRating = !ratedIds.has(messageId);
   474→    setRatedIds((prev) => new Set(prev).add(messageId));
   475→
   476→    try {
   477→      await patchJSON<{ ok: true }>("/api/tutor", { messageId, rating });
   478→      if (isFirstRating) {
   479→        toast({
   480→          title: "Gracias por tu feedback ⭐",
   481→          description: "Tu valoración nos ayuda a mejorar la tutoría.",
   482→        });
   483→      }
   484→    } catch {
   485→      // Revertir en caso de error
   486→      setMessages((prev) =>
   487→        prev.map((m) => (m.id === messageId ? { ...m, rating: null } : m))
   488→      );
   489→      setRatedIds((prev) => {
   490→        const next = new Set(prev);
   491→        next.delete(messageId);
   492→        return next;
   493→      });
   494→      toast({
   495→        title: "No se pudo guardar la calificación",
   496→        description: "Inténtalo nuevamente en unos segundos.",
   497→        variant: "destructive",
   498→      });
   499→    }
   500→  };
   501→
   502→  const handleRefresh = () => {
   503→    if (isSending) return;
   504→    refetchHistory();
   505→    toast({
   506→      title: "Conversación actualizada",
   507→      description: "Se ha recargado el historial desde el servidor.",
   508→    });
   509→  };
   510→
   511→  const userName = currentUser?.name ?? "Estudiante";
   512→  const hasMessages = messages.length > 0;
   513→
   514→  return (
   515→    <div className="mx-auto max-w-5xl space-y-6 p-4 lg:p-8">
   516→      <PageHeader
   517→        title="Tutor IA"
   518→        icon="MessageSquare"
   519→        iconGradient="from-violet-500 to-purple-600"
   520→        description="Tu asistente pedagógico. Te guía sin darte las respuestas — construye tu propio conocimiento."
   521→        actions={
   522→          <Button
   523→            variant="outline"
   524→            size="sm"
   525→            onClick={handleRefresh}
   526→            disabled={isSending || historyLoading}
   527→            className="gap-1.5"
   528→          >
   529→            <RefreshCw
   530→              className={cn("h-4 w-4", historyLoading && "animate-spin")}
   531→            />
   532→            <span className="hidden sm:inline">Regenerar</span>
   533→          </Button>
   534→        }
   535→      />
   536→
   537→      {/* Selector de contexto + disclaimer */}
   538→      <Card className="overflow-hidden">
   539→        <CardHeader className="pb-3">
   540→          <CardTitle className="flex items-center gap-2 text-base">
   541→            <BookOpen className="h-4 w-4 text-violet-500" />
   542→            ¿Sobre qué unidad quieres conversar?
   543→          </CardTitle>
   544→          <CardDescription>
   545→            Elige un tema para que el tutor ajuste sus preguntas. Puedes cambiar
   546→            en cualquier momento.
   547→          </CardDescription>
   548→        </CardHeader>
   549→        <CardContent className="space-y-4">
   550→          {/* Chips de unidades */}
   551→          <div className="-mx-1 flex gap-2 overflow-x-auto px-1 pb-1">
   552→            {/* Opción General */}
   553→            <button
   554→              type="button"
   555→              onClick={() => handleSelectContext(null)}
   556→              aria-pressed={selectedContext === null}
   557→              className={cn(
   558→                "flex shrink-0 items-center gap-2 rounded-full border px-3.5 py-1.5 text-sm font-medium transition",
   559→                selectedContext === null
   560→                  ? "border-transparent bg-gradient-to-br from-slate-600 to-slate-700 text-white shadow-md"
   561→                  : "border-border bg-card text-foreground/80 hover:border-violet-300 hover:bg-violet-50/50 dark:hover:border-violet-700 dark:hover:bg-violet-950/30"
   562→              )}
   563→            >
   564→              <Sparkles className="h-3.5 w-3.5" />
   565→              General
   566→            </button>
   567→
   568→            {unitsData === undefined &&
   569→              Array.from({ length: 4 }).map((_, i) => (
